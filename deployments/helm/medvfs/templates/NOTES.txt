━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  1. Check pods are running:

     kubectl get pods -l app.kubernetes.io/name={{ include "medvfs.name" . }}

  2. View the file manifest:

     kubectl port-forward svc/{{ include "medvfs.fullname" . }} {{ .Values.service.port }}:{{ .Values.service.port }}
     curl http://localhost:{{ .Values.service.port }}/manifest | jq .

  3. Check cache statistics:

     curl http://localhost:{{ .Values.service.port }}/stats

  4. Stream live events:

     curl -N http://localhost:{{ .Values.service.port }}/events
